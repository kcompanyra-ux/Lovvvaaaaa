function renderTickets(tickets) { const statusBadge = document.getElementById('statusBadge'); const headerCount = document.getElementById('headerTicketCount'); const grid = document.getElementById('ticketGrid'); const soldOutMsg = document.getElementById('soldOutMessage'); grid.innerHTML = ''; // Count ONLY available tickets for the header const availableCount = tickets.filter(t => t.status === 'available').length; headerCount.innerText = availableCount; if (availableCount === 0) { statusBadge.innerText = "0 tickets remaining"; statusBadge.parentElement.classList.replace('bg-secondary', 'bg-destructive'); statusBadge.parentElement.classList.replace('text-secondary-foreground', 'text-destructive-foreground'); soldOutMsg.classList.remove('hidden'); } else { statusBadge.innerText = `${availableCount} tickets available`; soldOutMsg.classList.add('hidden'); } // Sort tickets by ID so they always show 1 to 50 in perfect order tickets.sort((a, b) => a.id - b.id); tickets.forEach(ticket => { const btn = document.createElement('button'); if (ticket.status === 'available') { // Available ticket styling btn.className = 'min-h-12 rounded-lg bg-primary px-4 font-semibold text-primary-foreground shadow-md hover:bg-primary/90 hover:scale-105 transition-all flex items-center justify-center'; btn.innerText = `#${ticket.id}`; btn.onclick = () => openModal(ticket.id); } else { // Booked/Taken ticket styling btn.className = 'min-h-12 rounded-lg bg-muted border border-border px-4 font-semibold text-muted-foreground cursor-not-allowed opacity-70 flex flex-col items-center justify-center gap-1'; btn.disabled = true; btn.innerHTML = ` #${ticket.id} Booked `; } grid.appendChild(btn); }); }
